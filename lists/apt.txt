# Packages setup.sh makes sure are installed with apt. One package per line;
# anything after # is a comment.
#
# setup.sh installs what's missing, and removes a package dropped from this
# list only if setup.sh installed it (Ubuntu's own packages always stay).
# Keep it short: GUI apps come from Flathub and CLI tools from mise (both
# through chezmoi).

# Needed by setup.sh itself (Ubuntu's desktop doesn't include all of them)
curl
gpg                       # checks the vendor keys in apt-keys.txt
git
flatpak                   # Flathub apps (the dotfiles install them)

alacritty                 # the terminal (Super+Return, set up by the dotfiles)
ddcutil                   # ddc-brightness.sh; needs the host's /dev/i2c-*
xdg-terminal-exec         # runs terminal apps in the terminal from ~/.config/xdg-terminals.list (comes with Ubuntu; listed so it stays)
vim                       # the editor (EDITOR and Git's, set by the dotfiles); Ubuntu only has vim-tiny, with no vim command
gcc                       # C compiler; virtualbox-dkms builds its modules with it (dropping it would remove dkms and VirtualBox)
gh			  # official GitHub CLI tool
tmux                      # config in the dotfiles (~/.config/tmux/tmux.conf)
tree			  # recursively lists the contents of a directory in a tree-like visual format
htop                      # process viewer
btop                      # resource monitor (keeps its own default config)
rclone                    # mounts Google Drive (the dotfiles' rclone-smr.service)
gnome-tweaks              # GNOME settings that Settings doesn't show (no Flatpak exists)
distrobox                 # containers, for anything that doesn't belong on the host
podman                    # distrobox's container engine

# Virtual machines: QEMU/KVM through libvirt, run from virt-manager. setup.sh
# adds you to the libvirt group (groups.txt) and keeps libvirt's default NAT
# network starting at boot.
qemu-system-x86           # the emulator, using KVM
qemu-utils                # qemu-img, for disk images
libvirt-daemon-system     # libvirt's system daemon (qemu:///system)
libvirt-clients           # virsh
virt-manager              # Virtual Machine Manager, the GUI (native: the Flatpak can't see the host's QEMU)
virtinst                  # virt-install and virt-clone (a virt-manager dependency; listed so it stays)
ovmf                      # UEFI firmware for VMs

# VirtualBox, from Ubuntu's multiverse. Its kernel modules are built by
# virtualbox-dkms, which also sets `options kvm enable_virt_at_load=0`
# (/usr/lib/modprobe.d/virtualbox-dkms.conf): since kernel 6.12 KVM takes
# AMD-V/VT-x at boot and VirtualBox VMs won't start. With it, KVM only takes
# it while a KVM VM runs, so a VirtualBox VM and a KVM one (virt-manager,
# Claude Desktop's Cowork) can't run at the same time. Effective after a
# reboot. With Secure Boot on, the modules are signed with the machine's MOK
# key, which must be enrolled once per machine: the README's "Setting up a
# new machine", step 3.
virtualbox                # VBoxManage and the VM runtime
virtualbox-qt             # VirtualBox Manager, the GUI

# From the vendor repos in apt-sources/ (keys in apt-keys.txt)
brave-browser             # Brave, the default browser (set by the dotfiles)
code                      # Visual Studio Code
mise                      # installs the CLI tools in the dotfiles' ~/.config/mise/config.toml
claude-desktop            # Claude Desktop (its recommends add virtiofsd, for Cowork's VM)
